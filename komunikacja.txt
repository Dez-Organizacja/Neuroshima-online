dla frontendu:
{
    "view" : {
        "state" : {
            "fractions" : list[str] //obie frakcja
            "board" : list[list[format jednostki | None]] //lista 5 x 9
            "hands" : {frakcja : list[str] } // nazwy żetonów
        },
        "availableActions" : {
            "hand" : {frackja : [True | False] * 3},
            "board" : list[list[True | False]] //lista 5 x 9
        },
        "uiState" : {
            "fraction" : str // current frakcja
            "mode" : ("default"|"rotation"|"decision")
            "message" : str // widomość do wyświetlenia
        }
    }
}
format jednostki
{
    "name": str
    "fraction": str
    "ROTATION": str
    "DAMAGE": str
    "WIRED": str
}

dla engine:{
    "messageType" : ("newgame" / "user_action")
    "gameState" : cały poprzedni stan gry
    "userAction" : format akcji
}

format akcji:
{"type" : (hand/baord/bottom/rotate), ...}
{"type" : "hand", "slot" : (0/1/2)}
{"type" : "board", "pos": (int, int)}
{"type" : "rotate", "rotation" : int}
{"type" : "bottom", "name" : (
    "end_turn"/
    "discard"/
    "use"/
    "cancel"/
    "yes"/
    "no"/
    )
}
