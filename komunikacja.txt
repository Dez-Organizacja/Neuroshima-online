dla frontendu:
{
    "view" : {
        "state" : {
            "board" : list[5][9] = (True | False)
            "hands" : {frakcja : list[str] } // nazwy żetonów
        },
        "available_actions" : {
            "hand" : {frackja : [True | False] * 3},
            "board" : {(x, y) : True| False} // po wszystkich pozycjach
        },
        "ui_state" : {
            "fraction" : str // current frakcja
            "mode" : ("default"|"rotation"|"decision")
            "message" : str // widomość do wyświetlenia
        }
    }
}

dla engine:{
    "state" : cały poprzedni stan gry
    "type" : ("newgame" / "user_action")
    "action" : format akcji
}

format akcji:
{"type" : (hand/baord/bottom/rotate), ...}
{"type" : "hand", "slot" : (0/1/2)}
{"type" : "board", "pos": (int, int)}
{"type" : "rotate", "rotation" : int}
{"type" : "bottom", "name" : (
    "end_turn"/
    "discard"/
    "use"/
    "cancel"/
    "yes"/
    "no"/
    )
}
